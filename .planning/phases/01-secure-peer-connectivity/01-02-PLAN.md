1. **`parse_peer_address`** helper to parse `"<hex_pubkey>@<host>:<port>"` strings:
   ```rust
   /// Parse a peer address string of format "<hex_pubkey>@<host>:<port>"
   /// into an Ed25519 public key and socket address.
   fn parse_peer_address(addr: &str) -> Result<(ed25519::PublicKey, std::net::SocketAddr), AggregatorError> {
       let parts: Vec<&str> = addr.splitn(2, '@').collect();
       if parts.len() != 2 {
           return Err(AggregatorError::P2p(
               format!("Invalid peer address format '{}', expected '<hex_pubkey>@<host>:<port>'", addr)
           ));
       }
       let pubkey_bytes = const_hex::decode(parts[0])
           .map_err(|e| AggregatorError::P2p(format!("Invalid hex pubkey in '{}': {}", addr, e)))?;
       let pubkey = ed25519::PublicKey::try_from(pubkey_bytes.as_slice())
           .map_err(|e| AggregatorError::P2p(format!("Invalid Ed25519 pubkey in '{}': {}", addr, e)))?;
       let socket_addr: std::net::SocketAddr = parts[1].parse()
           .map_err(|e| AggregatorError::P2p(format!("Invalid socket address in '{}': {}", addr, e)))?;
       Ok((pubkey, socket_addr))
   }
   ```

3. **`spawn_commonware_runtime`** -- the core runtime scaffold:
   ```rust
   use commonware_runtime::tokio::{Config as RuntimeConfig, Runner};
   use commonware_p2p::authenticated::lookup;
   use commonware_p2p::Address;
   use governor;

   /// Spawn the commonware P2P runtime on a dedicated OS thread.
   ///
   /// Returns a JoinHandle for the OS thread and an mpsc sender for P2pCommands.
   ///
   /// CRITICAL: Must use std::thread::spawn (NOT tokio::spawn or spawn_blocking)
   /// because commonware's Runner creates its own Tokio runtime internally.
   /// Nesting Tokio runtimes causes a panic.
   fn spawn_commonware_runtime(
       private_key: ed25519::PrivateKey,
       p2p_config: P2pConfig,
       command_rx: mpsc::UnboundedReceiver<P2pCommand>,
   ) -> Result<std::thread::JoinHandle<()>, AggregatorError> {
       let handle = std::thread::spawn(move || {
           let runner = Runner::new(
               RuntimeConfig::default()
                   .with_worker_threads(2)
                   .with_max_blocking_threads(4)
                   .with_tcp_nodelay(Some(true))
           );
           runner.start(|context| async move {
               match p2p_config {
                   P2pConfig::Local { listen_port, ref peer_addresses, ref authorized_peers } => {
                       run_lookup_network(
                           context, &private_key, listen_port,
                           peer_addresses, authorized_peers, command_rx,
                       ).await;
                   }
                   P2pConfig::Remote { .. } => {
                       // Discovery mode -- implemented in Plan 03
                       tracing::warn!("Discovery mode not yet implemented");
                   }
                   P2pConfig::Disabled => {
                       // Should not reach here; handled before spawn
                   }
               }
           });
       });
       Ok(handle)
   }
   ```

4. **`run_lookup_network`** -- the lookup mode network lifecycle:
   ```rust
   /// Run a lookup-mode P2P network inside the commonware runtime.
   ///
   /// This function:
   /// 1. Creates a lookup::Network with the node's Ed25519 identity
   /// 2. Configures the Oracle with authorized peers + own pubkey
   /// 3. Registers a single broadcast channel (for Phase 2)
   /// 4. Starts the network
   /// 5. Runs a bridge loop handling P2pCommands from the WAVS main runtime
   ///
   /// SEC-02: Rate limiting MUST be active on all connections. The executor MUST
   /// verify that the constructed lookup::Config has non-zero rate-limiting fields.
   /// First try Config::local() defaults. If those do not set rate-limiting fields,
   /// explicitly set them via builder methods after construction (see fallback below).
   /// Either path is acceptable as long as rate limiting is confirmed active.
   async fn run_lookup_network(
       context: impl commonware_runtime::Spawner + commonware_runtime::Clock
           + commonware_runtime::Network + commonware_runtime::Metrics
           + commonware_runtime::BufferPooler + commonware_runtime::CryptoRngCore + Clone,
       private_key: &ed25519::PrivateKey,
       listen_port: u16,
       peer_addresses: &[String],
       authorized_peer_hexes: &[String],
       mut command_rx: mpsc::UnboundedReceiver<P2pCommand>,
   ) {
       let listen_addr = std::net::SocketAddr::from(([0, 0, 0, 0], listen_port));

       // Create lookup network config
       let config = lookup::Config::local(
           private_key.clone(),
           b"wavs-p2p",  // namespace for replay protection
           listen_addr,
           65536,  // max_message_size (64KB)
       );

       // SEC-02: Ensure rate limiting is active. The executor MUST inspect the
       // Config returned by Config::local() (via Debug print or docs.rs source).
       //
       // If Config::local() already populates these fields with non-zero defaults,
       // no further action needed -- just log confirmation.
       //
       // If Config::local() does NOT populate rate-limiting fields, apply explicit
       // rate limits unconditionally:
       //
       //   let config = config
       //       .with_allowed_connection_rate_per_peer(governor::Quota::per_second(
       //           std::num::NonZeroU32::new(10).unwrap()))
       //       .with_allowed_handshake_rate_per_ip(governor::Quota::per_second(
       //           std::num::NonZeroU32::new(5).unwrap()))
       //       .with_allowed_handshake_rate_per_subnet(governor::Quota::per_second(
       //           std::num::NonZeroU32::new(20).unwrap()));
       //
       // EXECUTOR MUST: Pick exactly one of the two paths above based on what
       // Config::local() actually returns, and DELETE the other path. Do NOT leave
       // both as commented code. The final code must have rate limiting provably
       // active (either via defaults or explicit calls).
       //
       // To verify: After constructing config, add a tracing::debug! that logs the
       // config fields, OR check the Config::local() source on docs.rs. If the
       // Config struct does not expose rate-limiting fields for inspection, apply
       // the explicit builder calls unconditionally (safe to set even if defaults
       // already exist -- explicit values override defaults).

       let (mut network, mut oracle) = lookup::Network::new(context.clone(), config);

       // Build Oracle peer map: Map<PublicKey, Address>
       // Include own pubkey (implicitly trusted per user decision)
       let own_pubkey = private_key.public_key();
       let mut peer_map = commonware_utils::Map::new();
       // Note: own pubkey is added to Oracle so other peers can verify us
       peer_map.insert(own_pubkey.clone(), Address::Network(listen_addr));

       // Parse and add known peer addresses
       for addr_str in peer_addresses {
           match parse_peer_address(addr_str) {
               Ok((pubkey, socket_addr)) => {
                   peer_map.insert(pubkey, Address::Network(socket_addr));
               }
               Err(e) => {
                   tracing::error!("Skipping invalid peer address '{}': {}", addr_str, e);
               }
           }
       }

       // Parse and add authorized peers (these may not have addresses in lookup mode,
       // but they need to be in the Oracle for authorization)
       // NOTE: In lookup mode, peers without addresses cannot be connected to,
       // but authorized_peers without addresses are still recognized if they connect to us.
       for hex_key in authorized_peer_hexes {
           match parse_authorized_peers(&[hex_key.clone()]) {
               Ok(keys) => {
                   for key in keys {
                       if !peer_map.contains_key(&key) {
                           // Add with Network address of 0.0.0.0:0 as placeholder
                           // (they connect to us, we don't need their address)
                           peer_map.insert(key, Address::Network(
                               std::net::SocketAddr::from(([0, 0, 0, 0], 0))
                           ));
                       }
                   }
               }
               Err(e) => {
                   tracing::error!("Skipping invalid authorized peer '{}': {}", hex_key, e);
               }
           }
       }

       // Register peer set with Oracle at index 0
       oracle.track(0, peer_map).await;

       tracing::info!(
           "P2P lookup network: listening on {}, {} peers configured",
           listen_addr, peer_addresses.len()
       );

       // Register a single broadcast channel (channel 0) for Phase 2 use
       let (_sender, _receiver) = network.register(
           0u32,
           governor::Quota::per_second(std::num::NonZeroU32::new(100).unwrap()),
           1024,
       );

       // Start the network
       let _net_handle = network.start();

       tracing::info!(
           "P2P network started (peer_id: {})",
           const_hex::encode(own_pubkey.as_ref())
       );

       // Bridge loop: handle P2pCommands from WAVS main runtime
       // Phase 1 only handles GetStatus; Publish/Subscribe/Unsubscribe are Phase 2
       loop {
           match command_rx.recv().await {
               Some(P2pCommand::GetStatus { response_tx }) => {
                   let status = P2pStatus {
                       peer_id: const_hex::encode(own_pubkey.as_ref()),
                       listen_addresses: vec![listen_addr.to_string()],
                       connected_peers: vec![], // Phase 2 fills this from network state
                       subscribed_topics: vec![], // Phase 2 fills this
                       mesh_peers: Default::default(), // Phase 2 fills this
                   };
                   let _ = response_tx.send(status);
               }
               Some(_) => {
                   // Publish, Subscribe, Unsubscribe, BlockPeer handled in Phase 1 Plan 03 / Phase 2
                   tracing::debug!("P2pCommand not yet implemented in Phase 1 Plan 02");
               }
               None => {
                   // Channel closed -- shutdown
                   tracing::info!("P2P command channel closed, shutting down network");
                   break;
               }
           }
       }
   }
   ```
