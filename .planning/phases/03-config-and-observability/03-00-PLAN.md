```rust
//! Wave 0 test stubs for P2pConfig (CFG-01 and CFG-02).
//! These will be filled in by Plan 03-01 when P2pConfig gets optional tuning fields.

#[cfg(test)]
mod tests {
    use super::super::p2p::P2pConfig;

    /// CFG-01: P2pConfig roundtrips through TOML correctly.
    /// Stub -- will be expanded in Plan 03-01 to cover max_message_size and deque_size.
    #[test]
    fn p2p_config_serde() {
        // Verify Disabled variant roundtrips
        let disabled = P2pConfig::Disabled;
        let toml_str = toml::to_string(&disabled).expect("serialize Disabled");
        let parsed: P2pConfig = toml::from_str(&toml_str).expect("deserialize Disabled");
        assert_eq!(parsed, P2pConfig::Disabled);

        // Verify Local variant roundtrips
        let local = P2pConfig::Local {
            listen_port: 9000,
            peer_addresses: vec!["abc123@127.0.0.1:9001".to_string()],
            authorized_peers: vec![],
        };
        let toml_str = toml::to_string(&local).expect("serialize Local");
        let parsed: P2pConfig = toml::from_str(&toml_str).expect("deserialize Local");
        assert_eq!(parsed, local);
    }

    /// CFG-02: Default values for optional tuning fields.
    /// Stub -- will be expanded in Plan 03-01 to verify max_message_size and deque_size defaults.
    #[test]
    fn p2p_config_defaults() {
        // Default is Disabled
        let default = P2pConfig::default();
        assert_eq!(default, P2pConfig::Disabled);

        // Local variant with only required fields should deserialize (optional fields default)
        let toml_str = r#"
            [local]
            listen_port = 9000
        "#;
        let parsed: P2pConfig = toml::from_str(toml_str).expect("deserialize Local with defaults");
        match parsed {
            P2pConfig::Local { listen_port, peer_addresses, authorized_peers, .. } => {
                assert_eq!(listen_port, 9000);
                assert!(peer_addresses.is_empty());
                assert!(authorized_peers.is_empty());
            }
            other => panic!("Expected Local, got {:?}", other),
        }
    }
}
```

Note: The `..` pattern in the match arm is used deliberately so that when Plan 03-01 adds `max_message_size` and `deque_size` fields, this test still compiles without modification. Plan 03-01 will then expand the tests to assert on the new fields.
