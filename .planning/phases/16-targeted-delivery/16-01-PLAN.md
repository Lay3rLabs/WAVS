// PeerSubscriptionMap -- lives as `peer_subscriptions` variable in both bridge loops

pub(crate) struct PeerSubscriptionMap { ... }

impl PeerSubscriptionMap {
    /// Get targeted recipient set for a service.
    /// Returns Recipients::Some(peers) if peers are known; Recipients::All as fallback (TGT-02).
    /// Unknown service_id or empty subscriber set both return Recipients::All.
    pub fn get_recipients(&self, service_id: &[u8; 32]) -> Recipients<ed25519::PublicKey>

    /// Returns true if peer has ever sent a subscription announcement (COMPAT-03).
    pub fn has_announced(&self, peer: &ed25519::PublicKey) -> bool

    /// Replace all subscriptions for a peer (replace-not-merge, used for full_state heartbeat).
    pub fn set_peer_subscriptions(
        &mut self,
        peer: &ed25519::PublicKey,
        service_ids: Vec<[u8; 32]>,
    )
}

// P2pMessage -- carries service_id_bytes for recipient resolution on retry drain
pub struct P2pMessage {
    pub service_id_bytes: [u8; 32],  // 32-byte service hash; SUBSCRIPTION_SENTINEL for announcements
    pub payload: Vec<u8>,
}

// Recipients enum from commonware-p2p
enum Recipients<P> {
    All,           // Deliver to all connected peers
    Some(Vec<P>),  // Deliver to specific peers only
    One(P),        // Deliver to exactly one peer (used for hello)
}
